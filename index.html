import time import network import machine import ubinascii import ujson from machine import Pin, time_pulse_us from umqtt.simple import MQTTClient, MQTTException import config trig = Pin(25, Pin.OUT, value=0) echo = Pin(26, Pin.IN) wifi = network.WLAN(network.STA_IF) wifi.active(True) CLIENT_ID = b"esp32-srf05-" + ubinascii.hexlify( machine.unique_id() ) def conectar_wifi(): if wifi.isconnected(): return print("Conectando Wi-Fi...") wifi.disconnect() wifi.connect(config.WIFI_SSID, config.WIFI_PASSWORD) inicio = time.ticks_ms() while not wifi.isconnected(): if time.ticks_diff(time.ticks_ms(), inicio) > 15000: wifi.disconnect() raise OSError("Tiempo de espera Wi-Fi agotado") time.sleep_ms(250) print("Wi-Fi conectado. IP:", wifi.ifconfig()[0]) def medir_distancia(): trig.value(0) time.sleep_us(2) trig.value(1) time.sleep_us(10) trig.value(0) duracion = time_pulse_us(echo, 1, 30000) # MicroPython devuelve un valor negativo si vence el plazo. if duracion <= 0: return None return round(duracion / 58.0, 1) def ejecutar(): while True: cliente = None try: conectar_wifi() cliente = MQTTClient( CLIENT_ID, config.MQTT_HOST, port=config.MQTT_PORT, user=config.MQTT_USER, password=config.MQTT_PASSWORD, keepalive=60, ) cliente.connect(timeout=5) print("MQTT conectado") while wifi.isconnected(): distancia = medir_distancia() mensaje = ujson.dumps({ "distancia_cm": distancia, "estado": "ok" if distancia is not None else "sin_eco", }) cliente.publish( config.MQTT_TOPIC, mensaje.encode(), retain=False, qos=0, ) print("Publicado:", mensaje) time.sleep(1) raise OSError("Se perdió la conexión Wi-Fi") except (OSError, MQTTException) as error: print("Error de conexión:", error) print("Reintentando en 5 segundos...") finally: if cliente is not None and cliente.sock is not None: try: cliente.sock.close() except OSError: pass time.sleep(5) ejecutar()
